138template <typename Function>
139auto MemoryScanner::submit(Function&& function)
140 -> std::future<std::invoke_result_t<std::decay_t<Function>&>> {
141 using Task = std::decay_t<Function>;
142 using Result = std::invoke_result_t<Task&>;
143
144 auto task = std::make_shared<std::packaged_task<Result()>>(
145 std::forward<Function>(function));
146
147 auto future = task->get_future();
148
149 boost::asio::post(m_pool, [task] { (*task)(); });
150
151 return future;
152}
